\begin{frame}{Hubbard model}
    \begin{equation}
        H
        =
        -t\sum_{\langle i j\rangle\sigma}
        (d^\dag_{i\sigma} d^{\pdag}_{j\sigma} + d^\dag_{j\sigma} d^{\pdag}_{i\sigma})
        +
        U\sum_i n_{i\uparrow} n_{i\downarrow}
        +
        (\epsilon - \mu)\sum_{i\sigma} n_{i\sigma}
    \end{equation}

\begin{frame}{Anderson impurity model}
    \begin{equation}
        H
        =
        % impurity
        U n_\uparrow n_\downarrow
        + (\epsilon - \mu) \sum_\sigma n_\sigma
        % bath
        + \sum_{k\sigma} \epsilon_k b^\dag_{k\sigma} b^{\pdag}_{k\sigma}
        % hybridization
        + \sum_{k\sigma} V^{\ps}_k d^\dag_{\pk\sigma} b^{\pdag}_{k\sigma}
        + V_k^* b^\dag_{k\sigma} d^{\pdag}_{\pk\sigma}
    \end{equation}
